class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> with TickerProviderStateMixin {
  // Keys
  final GlobalKey<ScaffoldState> _scaffoldKey = GlobalKey<ScaffoldState>();

  // Services
  final MarketService _marketService = MarketService();
  final WalletWinUpdater _walletWinUpdater = WalletWinUpdater();
  final WalletService _walletService = WalletService();
  final AuthService _authService = AuthService();
  final GameService _gameService = GameService();
  final BidService _bidService = BidService();
  final ResultService _resultService = ResultService();
  final Map<String, List<String>> panaCombinations = GameData.panaCombinations;
  final Map<String, List<String>> digitPanaMap = GameData.digitPanaMap;
  final List<String> triplePanaCombinations = GameData.triplePanaCombinations;

  // State variables
  double walletAmount = 0.0;
  String userName = '';
  String userNumber = '';
  String userId = '';
  bool _isRefreshing = false;
  bool _isProcessingWin = false;

  // Subscriptions
  late StreamSubscription<double> _walletSubscription;

  // Animations
  late AnimationController _fadeController;
  late AnimationController _walletPulseController;
  late Animation<double> _fadeAnim;
  late Animation<double> _walletPulseAnim;

  // Theme constants
  static const Color _dark = Color(0xFF0A1628);
  static const Color _surface = Color(0xFF162040);
  static const Color _cardBg = Color(0xFF1A2940);

  static const Color _goldLight = Color(0xFFF0C860);

  static const Color _red = Color(0xFFE74C3C);
  static const Color _textPrimary = Color(0xFFF0F4FF);
  static const Color _textSecondary = Color(0xFF8A9BB5);

  @override
  void initState() {
    super.initState();
    _initializeAnimations();
    _initializeWalletUpdater();
    _loadInitialData();
    _setupWalletUpdates();
    _checkForWinnings();
  }

  @override
  void dispose() {
    _fadeController.dispose();
    _walletPulseController.dispose();
    _walletSubscription.cancel();
    super.dispose();
  }

  // ==================== INITIALIZATION METHODS ====================

  void _initializeAnimations() {
    _fadeController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 700),
    );
    _walletPulseController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1500),
    )..repeat(reverse: true);
    _fadeAnim = CurvedAnimation(parent: _fadeController, curve: Curves.easeOut);
    _walletPulseAnim = Tween<double>(begin: 1.0, end: 1.04).animate(
      CurvedAnimation(parent: _walletPulseController, curve: Curves.easeInOut),
    );
  }

  Future<void> _initializeWalletUpdater() async {
    await _walletWinUpdater.initialize();
    _walletSubscription = _authService.walletUpdates.listen((balance) {
      if (mounted) setState(() => walletAmount = balance);
    });
  }

  Future<void> _loadInitialData() async {
    await _loadUserData();
    if (userId.isNotEmpty) await _refreshWalletBalance();
    _fadeController.forward();
  }

  Future<void> _loadUserData() async {
    try {
      final user = await _authService.getUserData();
      if (user != null && mounted) {
        setState(() {
          userName = user.name;
          userNumber = user.number;
          walletAmount = double.tryParse(user.walletAmount ?? '0.0') ?? 0.0;
          userId = user.id;
        });
      }
    } catch (e) {
      debugPrint('Error loading user data: $e');
    }
  }

  Future<void> _setupWalletUpdates() {
    _walletSubscription = _authService.walletUpdates.listen((balance) {
      if (mounted) setState(() => walletAmount = balance);
    });
    return Future.value();
  }

  // ==================== REFRESH METHODS ====================

  Future<void> _refreshData() async {
    if (_isRefreshing) return;
    setState(() => _isRefreshing = true);
    try {
      await Future.wait([
        _loadUserData(),
        if (userId.isNotEmpty) _refreshWalletBalance(),
        _checkForWinnings(),
      ]);
    } finally {
      if (mounted) setState(() => _isRefreshing = false);
    }
  }

  Future<void> _refreshWalletBalance() async {
    try {
      final walletResponse = await _walletService.getWallet(userId);
      if (mounted) setState(() => walletAmount = walletResponse.walletAmount);
      await _authService.refreshUserWallet();
    } catch (e) {
      debugPrint('Error refreshing wallet: $e');
    }
  }

  // ==================== WINNINGS CHECK METHODS ====================

  Future<void> _checkForWinnings() async {
    if (_isProcessingWin) return;
    setState(() => _isProcessingWin = true);

    try {
      final bids = await _bidService.getAllBids();
      final results = await _resultService.getAllResults();
      final gamesResponse = await _gameService.getDetailedGameList();

      final today = DateTime.now();
      final todayResults = _filterTodayResults(results, today);
      final userBids = bids.where((bid) => bid.userId == userId).toList();

      // Every bid is processed INDEPENDENTLY.
      // A user can win on ALL bids simultaneously — there is no short-circuit.
      for (final bid in userBids) {
        if (bid.win == "win" || bid.win == "loss") continue;
        await _processBidForWinnings(bid, todayResults, gamesResponse.games);
      }
    } catch (e) {
      debugPrint('Error checking for winnings: $e');
    } finally {
      if (mounted) setState(() => _isProcessingWin = false);
    }
  }

  List<Result> _filterTodayResults(List<Result> results, DateTime today) {
    return results.where((result) {
      DateTime? resultDate;
      try {
        resultDate = DateTime.parse(result.date.toString());
      } catch (_) {}
      if (resultDate == null) return false;
      return resultDate.year == today.year &&
          resultDate.month == today.month &&
          resultDate.day == today.day;
    }).toList();
  }

  Future<void> _processBidForWinnings(
    Bid bid,
    List<Result> todayResults,
    List<GameResponse> games,
  ) async {
    final result = todayResults.firstWhere(
      (r) => r.marketId == bid.marketId,
      orElse: () => Result.empty(),
    );
    if (result.id.isEmpty) return;

    final game = games.firstWhere(
      (g) => g.id == bid.gameId,
      orElse: () => GameResponse.empty(),
    );
    if (game.id.isEmpty) return;

    if (!_isResultReadyForBid(bid, result, game)) return;

    // Each bid is evaluated independently — win or loss has no effect on others
    final winInfo = _calculateWinnings(bid, result, game);

    if (winInfo['isWin'] as bool) {
      await _addWinningsToWallet(bid, winInfo['winAmount'] as double);
    } else {
      await _markBidAsLoss(bid);
    }
  }

  bool _isResultReadyForBid(Bid bid, Result result, GameResponse game) {
    if (bid.session == 'Open') {
      if (game.gameName == 'JODI DIGIT' ||
          game.gameName == 'FULL SANGAM' ||
          game.gameName == 'HALF SANGAM') {
        return result.closeResult.isNotEmpty && result.closePana.isNotEmpty;
      } else {
        return result.openResult.isNotEmpty || result.openPana.isNotEmpty;
      }
    } else if (bid.session == 'Close') {
      return result.closeResult.isNotEmpty || result.closePana.isNotEmpty;
    }
    return false;
  }

  // ==================== WIN CALCULATION ====================
  //
  // IMPORTANT: Every game type uses a standalone `if` (NOT `else if`).
  // This means ALL matching bids across ALL game types are evaluated
  // independently. A user who placed bids on multiple games can win
  // all of them simultaneously in the same session.

  Map<String, dynamic> _calculateWinnings(
    Bid bid,
    Result result,
    GameResponse game,
  ) {
    if (bid.marketId != result.marketId || bid.gameId != game.id) {
      return {'isWin': false, 'winAmount': 0.0};
    }
    if (bid.session == 'Open') {
      return _calculateOpenSessionWinnings(bid, result, game);
    } else if (bid.session == 'Close') {
      return _calculateCloseSessionWinnings(bid, result, game);
    }
    return {'isWin': false, 'winAmount': 0.0};
  }

  /// SP DP TP multiplier lookup — independent of other game checks.
  /// Triple Pana (600x) checked first since it could also appear in other maps.
  Map<String, dynamic> _calculateSpDpTpWinnings(Bid bid, String resultPana) {
    final pana = bid.pana.trim();
    if (pana != resultPana.trim()) return {'isWin': false, 'winAmount': 0.0};

    // Triple Pana → 600x
    if (GameData.triplePanaCombinations.contains(pana)) {
      return {'isWin': true, 'winAmount': double.parse(bid.points) * 600};
    }

    // Double Pana (digitPanaMap) → 300x
    final isDouble = GameData.digitPanaMap.values.any(
      (list) => list.contains(pana),
    );
    if (isDouble) {
      return {'isWin': true, 'winAmount': double.parse(bid.points) * 300};
    }

    // Single Pana (panaCombinations) → 150x
    final isSingle = GameData.panaCombinations.values.any(
      (list) => list.contains(pana),
    );
    if (isSingle) {
      return {'isWin': true, 'winAmount': double.parse(bid.points) * 150};
    }

    return {'isWin': false, 'winAmount': 0.0};
  }

  Map<String, dynamic> _calculateOpenSessionWinnings(
    Bid bid,
    Result result,
    GameResponse game,
  ) {
    final String gameName = game.gameName;
    final String bidDigit = bid.digit.trim();
    final String bidPana = bid.pana.trim();
    final String openResult = result.openResult.trim();
    final String closeResult = result.closeResult.trim();
    final String openPana = result.openPana.trim();
    final String closePana = result.closePana.trim();
    final double points = double.parse(bid.points);

    // Each game type returns immediately with its own result.
    // No `else if` chains — every game is fully independent.

    if (gameName == 'SINGLE DIGIT') {
      if (bidDigit == openResult) {
        return {'isWin': true, 'winAmount': points * 10};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'JODI DIGIT') {
      if (bidDigit == '$openResult$closeResult') {
        return {'isWin': true, 'winAmount': points * 100};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'RED JODI FAMILY') {
      if (bidDigit == '$openResult$closeResult') {
        return {'isWin': true, 'winAmount': points * 10};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'RED JODI') {
      if (bidDigit == '$openResult$closeResult') {
        return {'isWin': true, 'winAmount': points * 10};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'JODI FAMILY') {
      if (bidDigit == '$openResult$closeResult') {
        return {'isWin': true, 'winAmount': points * 10};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'SINGLE PANNA') {
      if (bidPana == openPana) {
        return {'isWin': true, 'winAmount': points * 150};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'DUBBLE PANNA') {
      if (bidPana == openPana) {
        return {'isWin': true, 'winAmount': points * 300};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'TRIPLE PANNA') {
      if (bidPana == openPana) {
        return {'isWin': true, 'winAmount': points * 600};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'CYCLE PATTI') {
      if (bidPana == openPana) {
        return {'isWin': true, 'winAmount': points * 140};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'FAMILY PANNA') {
      if (bidPana == openPana) {
        return {'isWin': true, 'winAmount': points * 150};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    // SP DP TP: auto-detects pana type and applies correct multiplier
    if (gameName == 'SP DP TP') {
      return _calculateSpDpTpWinnings(bid, openPana);
    }

    if (gameName == 'HALF SANGAM') {
      if (bidPana == '$openPana-$closeResult') {
        return {'isWin': true, 'winAmount': points * 1200};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'FULL SANGAM') {
      if (bidPana == '$openPana-$closePana') {
        return {'isWin': true, 'winAmount': points * 12000};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    return {'isWin': false, 'winAmount': 0.0};
  }

  Map<String, dynamic> _calculateCloseSessionWinnings(

    Result result,
    GameResponse game,
  ) {
    final String gameName = game.gameName;
    final String bidDigit = bid.digit.trim();
    final String bidPana = bid.pana.trim();
    final String openResult = result.openResult.trim();
    final String closeResult = result.closeResult.trim();
    final String closePana = result.closePana.trim();
    final double points = double.parse(bid.points);

    if (gameName == 'SINGLE DIGIT') {
      if (bidDigit == closeResult) {
        return {'isWin': true, 'winAmount': points * 10};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'SINGLE PANNA') {
      if (bidPana == closePana) {
        return {'isWin': true, 'winAmount': points * 150};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'DUBBLE PANNA') {
      if (bidPana == closePana) {
        return {'isWin': true, 'winAmount': points * 300};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'TRIPLE PANNA') {
      if (bidPana == closePana) {
        return {'isWin': true, 'winAmount': points * 600};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'CYCLE PATTI') {
      if (bidPana == closePana) {
        return {'isWin': true, 'winAmount': points * 140};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    if (gameName == 'FAMILY PANNA') {
      if (bidPana == closePana) {
        return {'isWin': true, 'winAmount': points * 150};
      }
      return {'isWin': false, 'winAmount': 0.0};
    }

    // SP DP TP: auto-detects pana type and applies correct multiplier
    if (gameName == 'SP DP TP') {

    }

    if (gameName == 'HALF SANGAM') {
      if (bidPana == '$openResult-$closePana') {

    }

    return {'isWin': false, 'winAmount': 0.0};
  }

  Future<void> _markBidAsLoss(Bid bid) async {
    try {
      await _bidService.updateBid(
        bid.id!,
        BidRequest(
          id: bid.id!,
          marketId: bid.marketId,
          userId: bid.userId,
          gameId: bid.gameId,
          name: bid.name,
          mobile: bid.mobile,
          market: bid.market,
          session: bid.session,
          digit: bid.digit,
          pana: bid.pana,
          points: bid.points,
          win: "loss",
          date: bid.date,
        ),
      );
    } catch (e) {
      debugPrint('Error marking bid as loss: $e');
    }
  }

  Future<void> _addWinningsToWallet(Bid bid, double amount) async {
    try {
      final newAmount = walletAmount + amount;

      final walletResponse = await http.put(
        Uri.parse('https://api.ekadantaa.in/api/user/$userId'),
        headers: {
          'Content-Type': 'application/json',
          'Authorization': 'Bearer ${await _authService.getToken()}',
        },
        body: json.encode({'walletAmount': newAmount.toStringAsFixed(2)}),
      );

      if (walletResponse.statusCode == 200) {
        await _bidService.updateBid(
          bid.id!,
          BidRequest(
            id: bid.id!,
            marketId: bid.marketId,
            userId: bid.userId,
            gameId: bid.gameId,
            name: bid.name,
            mobile: bid.mobile,
            market: bid.market,
            session: bid.session,
            digit: bid.digit,
            pana: bid.pana,
            points: bid.points,
            win: "win",
            date: bid.date,
          ),
        );

        await _authService.refreshUserData();

        if (mounted) {
          setState(() => walletAmount = newAmount);
          _showWinSnackbar(amount);
        }
      }
    } catch (e) {
      debugPrint('Error updating wallet: $e');
    }
  }

  void _showWinSnackbar(double amount) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Row(
          children: [
            const Icon(
              Icons.celebration_rounded,
              color: Colors.white,
              size: 18,
            ),
            const SizedBox(width: 8),
            Text(
              '🎉 ₹${amount.toStringAsFixed(2)} added to your wallet!',
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),
          ],
        ),
        backgroundColor: _green,
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        margin: const EdgeInsets.all(16),
        duration: const Duration(seconds: 4),
      ),
    );
  }

  // ==================== NAVIGATION METHODS ====================

  Future<void> _handleLogout() async {
    if (_scaffoldKey.currentState!.isDrawerOpen) Navigator.pop(context);

    final shouldLogout = await showDialog<bool>(
      context: context,
      builder: (context) => _buildLogoutDialog(),
    );

    if (shouldLogout == true) {
      final result = await _authService.logout();
      if (result['success'] == true && mounted) {
        Navigator.pushAndRemoveUntil(
          context,
          MaterialPageRoute(builder: (_) => const LoginScreen()),
          (r) => false,
        );
      }
    }
  }

  void _navigateToScreen(BuildContext context, Widget screen) {
    Navigator.pop(context);
    Navigator.push(context, MaterialPageRoute(builder: (_) => screen));
  }
